1Abstract

2Pipeline signature

Input

query(aoi: AOI, when?: Interval, families?: Family[], licences?: Licence[]) → FeatureSet

AOI      = GeoJSON.Polygon | GeoJSON.MultiPolygon      // EPSG:4326
Interval = { start: Date, end: Date }                   // default: latest
Family   = "socio-economic" | "infrastructure" | "hazards" | "macrobiology"
         | "geology" | "health-microbiology" | "climate" | "spectral"
         | "water" | "latent"                           // default: all
Licence  = "permissive" | "share-alike" | "noncommercial" | "restricted"

Output

FeatureSet   = { aoi: AOI, queried_at: DateTime, results: Result[] }

Result       = Found | Unavailable | NotAttempted      // one per layer × source
Found        = Slot & { status: "found", rank: int, fill: Fill,
                        observed: Date, res_m: number | "vector", features: Feature[] }
Unavailable  = Slot & { status: "unavailable", reason: UnavailableReason }
NotAttempted = Slot & { status: "not-attempted", reason: SkipReason }

Slot         = { layer: LayerId, source: SourceId, family: Family, licence: Licence }
Fill         = "observed" | "inferred" | "analogue"
Feature      = { name: FeatureId, value: number | string, unit?: string }

UnavailableReason = "no-data" | "source-error" | "auth-required"
SkipReason        = "out-of-coverage" | "family-excluded" | "licence-excluded"

4Method

metasmith plan→Nextflow driver→1 task per source × AOI
